# =============================================================================
# Qwen audio variants — paste-ready commands for all 4 Tasks
# =============================================================================
#
# Conventions:
#   * Run from the repo root.
#   * Output for each variant is written to data/output/<variant_tag>/
#     with subfolders Task1, Task2, Task3, Task4, internals/
#   * The original 4-bit pipeline (data/output/qwen/Task1..4) is untouched —
#     run_qwen_variant.py never writes there unless you set
#     --variant-tag qwen.
#   * `python` here means whatever interpreter you have activated. On Lambda
#     Labs Linux that is typically `python` (after `source venv/bin/activate`);
#     on local Windows replace with `.\.venv\Scripts\python.exe` if needed.
#
# Each section below is one self-contained variant. Pick the section you want
# and paste the command. To run several variants in sequence just paste them
# one after another — the runner is idempotent per output dir.
#
# =============================================================================
# Prereqs (Linux, Lambda Labs A10 24GB recommended for the fp16 7B variants)
# =============================================================================
#
#   git clone <repo> && cd audio-focus-inference-research
#   python -m venv .venv && source .venv/bin/activate
#   pip install -U pip
#   pip install -r requirements.txt
#   pip install -U "transformers>=4.50" accelerate bitsandbytes \
#                  soundfile librosa scikit-learn pandas
#
# (Qwen2.5-Omni requires transformers>=4.50; bitsandbytes is only needed for
#  4bit/8bit precision.)
#
#
# =============================================================================
# 0. Sanity check — re-run the original 4-bit Qwen2-Audio pipeline
#    (verifies the refactor matches your previous results)

python code/qwen/run_qwen_variant.py \

    --files        f1 f2 f3 f4 f5 f6 f7 f8 f9 f10


# =============================================================================
# 2. Qwen2.5-Omni-7B at fp16 (newer multimodal architecture)

# =============================================================================
python code/qwen/run_qwen_variant.py \
    --model        Qwen/Qwen2.5-Omni-7B \
    --precision    fp16 \
    --variant-tag  qwen2_5_omni_fp16 \
    --speaker      speaker0 \
    --files        f1 f2 f3 f4 f5 f6 f7 f8 f9 f10


# =============================================================================
# 3. Optional: include cross-speaker held-out probe data (speaker1, speaker2)

python code/qwen/run_qwen_variant.py \
    --model           Qwen/Qwen2-Audio-7B-Instruct \
